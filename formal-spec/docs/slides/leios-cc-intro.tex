\section{Protocol recap}
% ===========================================================================

\begin{frame}{Why Linear Leios?}
  \begin{itemize}
    \item Ouroboros Praos: in every slot, each stake pool privately checks
          whether it may produce a block
    \item A block must diffuse before the next one is likely to be
          produced, so throughput is bounded by the block size that
          diffuses in time, not by bandwidth or CPU.
    \item Linear Leios (CIP-0164) lifts that bound:
      \begin{itemize}
        \item \textcolor{rb}{ranking blocks} (RBs) keep the Praos chain and
              its security argument, and each announces at most one EB;
        \item large \textcolor{eb}{endorser blocks} (EBs) carry the
              transactions, diffused off the critical path;
        \item a stake-based committee \textcolor{vt}{votes} on an EB, and a
              \emph{certificate} in the next RB makes it part of the ledger.
      \end{itemize}
  \end{itemize}
\end{frame}

\section{Categorical Crypto refresher}
% ===========================================================================

\begin{frame}{Categorical Crypto}
  \begin{itemize}
    \item A compositional framework for interactive systems and their
          security proofs, including universal composability (UC), built on
          category theory and mechanised in Agda (\agda{--safe}).
    \item In the concrete model, typed \alert{channels} are the objects and
          interactive \alert{machines} are the morphisms; machines compose
          sequentially and in parallel.
    \item Protocols, functionalities, adversaries and environments are
          modelled as machines, with distinct roles in the security argument.
    \item The abstract UC layer proves reflexivity, transitivity, the dummy
          adversary reduction and a composition theorem, under its stated
          assumptions on composition and observational equivalence.
    \item Leios uses machine composition to build the specification and
          transfer safety and liveness; UC emulation is the separate goal for
          replacing ideal voting with real voting.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Channels}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
\begin{minted}{agda}
record Channel : Type₁ where
  constructor _⇿_
  field inType outType : Type

I : Channel
I = ⊥ ⇿ ⊥

_ᵀ : Channel → Channel
A ᵀ = A .outType ⇿ A .inType

_⊗₀_ : Channel → Channel → Channel
A ⊗₀ B = (inType A ⊎ inType B)
       ⇿ (outType A ⊎ outType B)
\end{minted}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{itemize}
        \item A channel is a pair of message types, one per direction.
        \item \agda{ᵀ} swaps the directions: the other end's view.
        \item \agda{⊗₀} bundles channels side by side; \agda{I} is its unit.
      \end{itemize}

      \vspace{3mm}
      \centering

\begin{frame}[fragile]{Machines}
\begin{minted}{agda}
machine-type : Type → Channel → Type₁
machine-type State C = State → inType C → Maybe (outType C) → State → Type

record Machine (A B : Channel) : Type₁ where
  field {State} : Type
        stepRel : machine-type State (A ⊗₀ B ᵀ)
\end{minted}
  \begin{center}
    % stepRel ranges over A ⊗₀ Bᵀ: the machine reads inType A and
    % outType B, and writes outType A and inType B.
    \begin{tikzpicture}[msg/.style={thick, -{Latex[length=2mm]}}]
      \node[machine, minimum width=16mm, minimum height=14mm] (m) {\agda{M}\\[-1mm]{\scriptsize State}};
      \draw[msg] ([xshift=-17mm, yshift=3mm]m.west)
        -- node[above, lbl]{inType A} ([yshift=3mm]m.west);
      \draw[msg] ([yshift=-3mm]m.west)
        -- node[below, lbl]{outType A} ([xshift=-17mm, yshift=-3mm]m.west);
      \draw[msg] ([yshift=3mm]m.east)
        -- node[above, lbl]{inType B} ([xshift=17mm, yshift=3mm]m.east);
      \draw[msg] ([xshift=17mm, yshift=-3mm]m.east)
        -- node[below, lbl]{outType B} ([yshift=-3mm]m.east);
      \node[left] at ([xshift=-17mm]m.west) {\agda{A}};
      \node[right] at ([xshift=17mm]m.east) {\agda{B}};
    \end{tikzpicture}
  \end{center}
  \begin{itemize}
    \item A machine from \agda{A} to \agda{B}: a relation that consumes one
          message and emits at most one.
    \item Relational, so nondeterminism is free: the spec says what an
          honest node \emph{may} do.
    \item Linear Leios is exactly such a relation:
          \agda{LinearLeios .stepRel = \_-⟦\_/\_⟧⇀\_}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{The node as a state machine}
  \begin{itemize}
    \item One relation \agda{\_-⟦\_/\_⟧⇀\_}: state, one input message,
          at most one output message, new state.
  \end{itemize}
\begin{minted}{agda}
data _-⟦_/_⟧⇀_ : machine-type LeiosState (((FFD ⊗₀ BaseIO) ⊗₀ VotingC) ⊗₀ (IO ⊗₀ Adv) ᵀ)
\end{minted}
  \begin{itemize}
    \item Every slot a node must settle four duties before time advances
          (\agda{allDone}):
      \begin{itemize}
        \item \agda{EB-Role}: propose an EB, or decline;
        \item \agda{VT-Role}: vote, decline, or postpone;
        \item \agda{CertCheck}: decide whether the RB carries a certificate;
        \item \agda{Base}: submit the RB.
      \end{itemize}
    \item A role is settled by performing it, \emph{provably} declining it,
          or, for voting, deferring while the window is open.
    \item An enabled EB role cannot be skipped; an eligible vote cannot
          be deferred beyond its deadline.
  \end{itemize}
\end{frame}

  \begin{itemize}
    \item Premises are decidable; \agda{genPremises} derives the decision
          procedures used by the executable trace verifier.
  \end{itemize}

\begin{frame}[fragile]{Rule \agda{Slot}: advance the slot}
\begin{minted}{agda}
Slot : let open LeiosState s in
      ∙ allDone s
      ──────────────────────────────────────────────────────────────────
      s -⟦ ((ϵ ⊗R) ⊗R) ⊗R ↑ᵢ FFD-OUT msgs / just $ ((L⊗ ϵ) ⊗R) ⊗R ↑ₒ FTCH-LDG ⟧⇀
        let s' = s ↑ L.filter (isValid? s) msgs
        in record s'
             { slot       = suc slot
             ; Upkeep     = []
             ; proposedEB = nothing
             }
\end{minted}
  \begin{ruleexplanation}{FFD}{BaseIO}
    \item Fires only once all four duties of the slot are settled
          (\agda{allDone}).
    \item Takes in the valid EBs and headers the network delivered, asks
          for the ledger, and resets the duties for the next slot.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{Chain}: track the base chain}
\begin{minted}{agda}
Chain : let open LeiosState s in
      ───────────────────────────────────────────────────────────────────
      s -⟦ ((L⊗ ϵ) ⊗R) ⊗R ↑ᵢ BASE-LDG rbs / nothing ⟧⇀ record s { RBs = rbs }
\end{minted}
  \begin{ruleexplanation}{BaseIO}{none}
    \item The base chain reports its current blocks; the node records them.
    \item No premises: the tip can change at any moment, including between
          a certificate query and its answer (hence \agda{Cert₂},
          \agda{Cert₃}).
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{Fetch}: report the ledger}
\begin{minted}{agda}
Fetch : let open LeiosState s in
      ───────────────────────────────────────────────────────────────────────────
      s -⟦ L⊗ (ϵ ⊗R) ᵗ¹ ↑ₒ FetchLdgI / just $ L⊗ (ϵ ⊗R) ᵗ¹ ↑ᵢ FetchLdgO Ledger ⟧⇀ s
\end{minted}
  \begin{ruleexplanation}{IO}{IO}
    \item Answers a request for the node's view of the ledger.
    \item Enabled in every state, and leaves the state unchanged.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{Mempool}: replace the transaction selection}
\begin{minted}{agda}
Mempool :
        ───────────────────────────────────────────────────────────────────────────
        s -⟦ L⊗ (ϵ ᵗ¹ ⊗R) ᵗ¹ ↑ᵢ SubmitTxs txs / nothing ⟧⇀ record s { ToPropose = txs }
\end{minted}
  \begin{ruleexplanation}{IO}{none}
    \item The environment hands the node its current selection of
          transactions, which replaces \agda{ToPropose}.
    \item \agda{EB-Role} takes the EB share of it, and the rules that
          submit an RB take the RB share.
    \item No rule removes the transactions they include, so the
          environment must not offer them again.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{Base₁}: submit an RB without a certificate}
\begin{minted}{agda}
Base₁   : let open LeiosState s in
        ∙ needsUpkeep Base
        ∙ needsUpkeep CertCheck
        ∙ hasUpkeep EB-Role
        ∙ certRequest s ≡ nothing
        ───────────────────────────────────────────────────────────────────────────
        s -⟦ ((ϵ ⊗R) ⊗R) ⊗R ↑ᵢ SLOT / just $ ((L⊗ ϵ) ⊗R) ⊗R ↑ₒ SUBMIT (mkRB s nothing) ⟧⇀
          addUpkeep (addUpkeep s CertCheck) Base
\end{minted}
  \begin{ruleexplanation}{FFD}{BaseIO}
    \item The chain tip calls for no certificate, so the RB carries
          transactions; this settles both \agda{CertCheck} and \agda{Base}.
    \item \agda{hasUpkeep EB-Role}: the EB role is settled first, so the RB
          announces the EB the node diffuses (\agda{proposedEB}).
    \item The base chain only takes the RB into account if the node is its
          block producer for the slot.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{Base₂}: query for a certificate}
\begin{minted}{agda}
Base₂   : let open LeiosState s in
        ∙ needsUpkeep CertCheck
        ∙ hasUpkeep EB-Role
        ∙ certRequest s ≡ just eb
        ───────────────────────────────────────────────────────────────────────────
        s -⟦ ((ϵ ⊗R) ⊗R) ⊗R ↑ᵢ SLOT / just $ (L⊗ ϵ) ⊗R ↑ₒ QUERY (hash currentRB) ⟧⇀
          record (addUpkeep s CertCheck) { PendingQuery = just (hash currentRB) }
\end{minted}
  \begin{ruleexplanation}{FFD}{VotingC}
    \item The tip announces an EB whose voting window closed at least
          $L_\mathrm{diff}$ slots ago, so the node asks the voting
          functionality for a certificate.
    \item The query is keyed by the announcing RB, and recorded in
          \agda{PendingQuery}.
    \item \agda{Base} stays open until the answer arrives.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{Cert₁}: embed the certificate}
\begin{minted}{agda}
Cert₁ : let open LeiosState s in
      ∙ needsUpkeep Base
      ∙ CertCheck ∈ˡ Upkeep
      ∙ certRequest s ≡ just eb
      ∙ PendingQuery ≡ just (hash currentRB)
      ∙ Maybe.All (λ c → getEBHash c ≡ hash currentRB) c
      ───────────────────────────────────────────────────────────────────
      s -⟦ (L⊗ ϵ) ⊗R ↑ᵢ CERT c / just $ ((L⊗ ϵ) ⊗R) ⊗R ↑ₒ SUBMIT (mkRB s c) ⟧⇀
        record (addUpkeep s Base) { PendingQuery = nothing }
\end{minted}
  \begin{ruleexplanation}{VotingC}{BaseIO}
    \item Accepts an answer only while a query is outstanding, and only if
          the query still names the \emph{current} tip.
    \item A positive answer must certify the requested RB; a negative one
          (\agda{c = nothing}) yields an RB without a certificate.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{Cert₂}: the tip no longer needs a certificate}
\begin{minted}{agda}
Cert₂ : let open LeiosState s in
      ∙ needsUpkeep Base
      ∙ CertCheck ∈ˡ Upkeep
      ∙ certRequest s ≡ nothing
      ∙ PendingQuery ≡ just r
      ───────────────────────────────────────────────────────────────────
      s -⟦ (L⊗ ϵ) ⊗R ↑ᵢ CERT c / just $ ((L⊗ ϵ) ⊗R) ⊗R ↑ₒ SUBMIT (mkRB s nothing) ⟧⇀
        record (addUpkeep s Base) { PendingQuery = nothing }
\end{minted}
  \begin{ruleexplanation}{VotingC}{BaseIO}
    \item The tip changed while the query was outstanding, and the new tip
          calls for no certificate at all.
    \item The stale answer is discarded and the RB is submitted without a
          certificate, settling \agda{Base} as \agda{Base₁} would.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{Cert₃}: the tip moved, query again}
\begin{minted}{agda}
Cert₃ : let open LeiosState s in
      ∙ needsUpkeep Base
      ∙ CertCheck ∈ˡ Upkeep
      ∙ certRequest s ≡ just eb
      ∙ PendingQuery ≡ just r
      ∙ hash currentRB ≢ r
      ───────────────────────────────────────────────────────────────────
      s -⟦ (L⊗ ϵ) ⊗R ↑ᵢ CERT c / just $ (L⊗ ϵ) ⊗R ↑ₒ QUERY (hash currentRB) ⟧⇀
        record s { PendingQuery = just (hash currentRB) }
\end{minted}
  \begin{ruleexplanation}{VotingC}{VotingC}
    \item The new tip calls for a certificate on a \emph{different} RB
          than the one queried.
    \item The stale answer is discarded and a fresh query is issued;
          \agda{Base} stays open until that one is answered.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{EB-Role}: propose an EB}
\begin{minted}[fontsize=\scriptsize]{agda}
EB-Role : let open LeiosState s in
          ∙ CanProposeEB s π eb
          ∙ needsUpkeep EB-Role
          ──────────────────────────────────────────────────
          s -⟦ ((ϵ ⊗R) ⊗R) ⊗R ↑ᵢ SLOT / just $ ((ϵ ⊗R) ⊗R) ⊗R ↑ₒ FFD-IN (Send (ebHeader eb) nothing) ⟧⇀
            rememberProposal (addUpkeep s EB-Role) eb
\end{minted}
\begin{minted}[fontsize=\scriptsize]{agda}
CanProposeEB : LeiosState → VrfPf → EndorserBlock → Type
CanProposeEB s π eb = let open LeiosState s in
  toProposeEB s π ≡ just eb × canProduceEB slot sk-EB (stake s) π
\end{minted}
  \begin{ruleexplanation}{FFD}{FFD}
    \item \agda{canProduceEB} holds if and only if the node can make an RB
          in this slot; the EB holds the EB share of \agda{ToPropose}, if
          that share is non-empty.
    \item The EB goes to the network, and \agda{proposedEB} records it for
          the RB to announce.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{No-EB-Role}: decline to propose}
\begin{minted}{agda}
No-EB-Role : let open LeiosState s in
             ∙ ¬ (∃₂ λ π eb → CanProposeEB s π eb)
             ∙ needsUpkeep EB-Role
             ──────────────────────────────────────────────────
             s -⟦ ((ϵ ⊗R) ⊗R) ⊗R ↑ᵢ SLOT / nothing ⟧⇀ addUpkeep s EB-Role
\end{minted}
  \begin{ruleexplanation}{FFD}{none}
    \item The role is settled without an EB, but only if \emph{no} choice
          of proof and EB would let the node propose one.
    \item So an honest node cannot silently skip an EB it could have
          produced.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{VT-Role}: cast a vote}
\begin{minted}[fontsize=\scriptsize]{agda}
VT-Role : ∀ {ebHash slot'} → let open LeiosState s in
        ∙ CanVote s eb ebHash slot'
        ∙ needsUpkeep VT-Role
        ──────────────────────────────────────────────────
        s -⟦ ((ϵ ⊗R) ⊗R) ⊗R ↑ᵢ SLOT / just $ (L⊗ ϵ) ⊗R ↑ₒ CAST (vote sk-VT (hash currentRB)) ⟧⇀
          rememberVote (addUpkeep s VT-Role) eb
\end{minted}
  \begin{columns}[c]
    \begin{column}{0.30\textwidth}
      \centering\rulechannels{FFD}{VotingC}
    \end{column}
    \begin{column}{0.68\textwidth}
\begin{minted}[fontsize=\scriptsize]{agda}
CanVote : LeiosState → EndorserBlock → EBRef → ℕ → Type
CanVote s eb ebHash slot' = let open LeiosState s in
    getCurrentEBHash s ≡ just ebHash
  × find (λ (_ , eb') → hash eb' ≟ ebHash) EBs' ≡ just (slot' , eb)
  × hash eb ∉ VotedEBs
  × ¬ isEquivocated s eb
  × isValid s (inj₁ (ebHeader eb))
  × slot' ≤ slotNumber eb + Lhdr
  × slotNumber eb + 3 * Lhdr ≤ slot
  × slot ≤ slotNumber eb + 3 * Lhdr + Lvote
  × isValidityChecked slot eb
  × EndorserBlockOSig.txs eb ≢ []
  × inVotingCommittee params (stake s)
  × id ∈ˡ L.map poolID PubKeys
\end{minted}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Rule \agda{No-VT-Role}: decline to vote}
\begin{minted}{agda}
No-VT-Role : let open LeiosState s in
             ∙ ¬ (∃[ eb ] ∃[ ebHash ] ∃[ slot' ] CanVote s eb ebHash slot')
             ∙ needsUpkeep VT-Role
             ──────────────────────────────────────────────────
             s -⟦ ((ϵ ⊗R) ⊗R) ⊗R ↑ᵢ SLOT / nothing ⟧⇀ addUpkeep s VT-Role
\end{minted}
  \begin{ruleexplanation}{FFD}{none}
    \item The role is settled without a vote, but only if there is
          \emph{nothing} the node could vote on.
  \end{ruleexplanation}
\end{frame}

\begin{frame}[fragile]{Rule \agda{VT-Defer}: postpone a vote}
\begin{minted}{agda}
VT-Defer : let open LeiosState s in
         ∙ slot < voteDeadline s
         ∙ needsUpkeep VT-Role
         ──────────────────────────────────────────────────
         s -⟦ ((ϵ ⊗R) ⊗R) ⊗R ↑ᵢ SLOT / nothing ⟧⇀ addUpkeep s VT-Role
\end{minted}
  \begin{ruleexplanation}{FFD}{none}
    \item A node may abstain while the current EB's voting window is still
          open, even when \agda{VT-Role} could fire.
    \item At the deadline \agda{VT-Defer} no longer applies, and
          \agda{No-VT-Role} does not apply while a vote can still be cast:
          a node that can vote must vote by then.
  \end{ruleexplanation}
\end{frame}

% Leios/Linear/Trace/Verifier.lagda.md: Action, TestInput, TestTrace.
\begin{frame}[fragile]{From a recorded trace to specification steps}
\begin{minted}{agda}
TestInput = FFDT Out ⊎ BaseIOF In ⊎ IOT In ⊎ VotingT In
TestTrace = List (Action × TestInput)
\end{minted}
  \begin{itemize}
    \item Each entry names the expected rule and its slot, paired with the
          input consumed: for example, \agda{Base₂-Action 108} with
          \agda{SLOT}, or \agda{Cert₁-Action 108} with \agda{CERT (just c)}.
    \item Role actions also carry witnesses: \agda{EB-Role-Action} names
          the proposed EB; \agda{VT-Role-Action} names the EB and its recorded
          receipt slot.
    \item Given an initial \agda{LeiosState}, replay the actions against
          the \emph{same transition relation} that defines \agda{LinearLeios}.
          Outputs and successor states are reconstructed, not supplied by
          the trace.
    \item The internal list is newest first; the sample tests reverse
          chronological event lists before calling \agda{verifyTrace}.
    \item This is node-level conformance checking.  The external trace
          verifier uses this Agda checker for simulator and live-node traces.
  \end{itemize}
\end{frame}

% Leios/Linear/Trace/Verifier.lagda.md: ValidStep, ValidTrace, verifyStep.
\begin{frame}[fragile]{Verification returns a proof, not just a Boolean}
\begin{minted}{agda}
verifyStep : (a : Action) → (i : TestInput) → (s : LeiosState)
  → Result (Err-verifyStep a i s) (ValidStep (a , i) s)

verifyTrace : (σs : TestTrace) → (s : LeiosState)
  → Result (Err-verifyTrace σs s) (ValidTrace σs s)
\end{minted}
  \begin{enumerate}\small
    \item Check that the action's slot matches the current state and that
          the input constructor matches the rule.
    \item Decide the rule's premises, using decision procedures derived
          from the specification by \agda{genPremises}.
    \item On success, construct the actual transition and carry its
          successor state into the next check; on failure, return a
          structured error with the failed obligation.
  \end{enumerate}
  \begin{itemize}\small
    \item \agda{ValidTrace} contains a derivation of the execution and a
          proof that its actions and inputs match the recorded trace.
          \agda{getNewState} extracts its final state.
    \item Acceptance therefore witnesses a valid specification execution;
          it does not establish safety or liveness of every execution.
  \end{itemize}
\end{frame}

% Live reporting and checkpoint policy in linear-chain/Main.hs and
% LinearLeiosChain.hs; these are more detailed than the README's summary.
\begin{frame}[fragile]{Reading the live verifier's output}
\begin{minted}{text}
ok @ <events> events, <actions> actions
VIOLATION after <events> events: <status>
\end{minted}
  \begin{itemize}\small
    \item \textbf{Checkpoints}: every 20 slots, at epoch boundaries and at
          end of input, re-check the accumulated segment.  A violation
          reports the failing action and details, then exits unsuccessfully.
    \item Only completed slots are adjudicated; the final slot still in
          progress is left unverified when the stream ends.
    \item \textbf{Eligibility source}: the report says whether it used the
          queried leadership schedule or the node's own leadership log.
          A schedule for another epoch, or a segment spanning epochs, causes
          fallback to the log.

          obligations across those gaps are not checked.  Missing or
          inconclusive mempool readings can prevent enforcement of EB
          production; votes beyond the first in a slot are reported as unchecked.
    \item Interpret \texttt{ok} together with these notes: it certifies
          the translated, checked portion of the log, rather than every
          internal operation of the live node.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{The voting interface}
\begin{minted}{agda}
data VotingT : Mode → Type where
  CAST  : Vote → VotingT Out
  QUERY : EBRef → VotingT Out
  CERT  : Maybe EBCert → VotingT In

VotingC = simpleChannel VotingT
\end{minted}

    \item A certificate is assembled at RB production and travels inside
          the RB, rather than as a diffusion event to the node.
    \item Two functionalities serve this channel: a \alert{global}
          \agda{Certifier} and a \alert{local} \agda{Voter}.
  \end{itemize}
\end{frame}

          validating voter behind it.

\begin{frame}[fragile]{Certificate correctness: an honest validator exists}
\begin{minted}[fontsize=\scriptsize]{agda}
cert-correct : ∀ {st x}
  → WF st
  → (corrupt : List Party)
  → (∀ {p} → Voted p x st → ¬ honest p → p ∈ˡ corrupt)
  → length corrupt N.< threshold
  → Certified st x
  → ∃[ p ] (honest p × Validated p x)
\end{minted}
  \begin{itemize}\small
    \item \textbf{Invariant}: \agda{WF} says every logged vote by an honest
          party is backed by validation.  It holds for the empty log and is
          preserved by \agda{Cast}.
    \item \textbf{Counting argument}: a certificate contains at least
          \agda{threshold} distinct voters.  They cannot all belong to the
          smaller corrupt set, so one is honest; \agda{WF} gives validation.
    \item \textbf{Real voting}: filter the local log to \agda{Valid} votes
          and map each to its voter and reference.  A real quorum becomes an
          ideal quorum; \agda{real-cert-correct} transfers the result, and
          \agda{answered-cert-correct} applies it to a positive voter reply.
    \item The real result assumes that valid votes attributed to honest
          parties imply validation.  This certificate theorem is proved;
          full real-to-ideal UC emulation remains a separate goal.
  \end{itemize}
\end{frame}

\begin{frame}{UC emulation}
  \begin{itemize}
    \item Environments are tests: machines into a fixed output channel
          \agda{ℰ-Out}, possibly with an ancilla wire.
    \item \agda{f ≈ℰ g}: every test yields equivalent composites, i.e.\ no
          environment distinguishes \agda{f} from \agda{g}.
    \item \agda{π ≤UC φ}: for every adversary against the real protocol
          \agda{π} there is a simulator against the ideal functionality
          \agda{φ} such that the two worlds are \agda{≈ℰ}.
    \item The abstract layer proves, once, the following:
      \begin{itemize}
        \item \agda{≤UC} is a preorder;
        \item the dummy adversary is complete;
        \item \alert{composition}: if \agda{π ≤UC φ} then
              \agda{ρ ∘ π ≤UC ρ ∘ φ}.
      \end{itemize}
    \item Practical consequence: reason about Leios against an
          \emph{ideal} voting functionality, then replace it.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Safety: common prefix}
\begin{minted}{agda}
safeState : {A : Channel} → ℕ → (E : Environment A) → Machine.State (protocol E) → Type
safeState k E S =
  {p p' : Fin n} (honest-p : p ∈ honest-nodes) (honest-p' : p' ∈ honest-nodes)
  → prune k (getChain E S honest-p) ≼ getChain E S honest-p'

safety : ℕ → Type₁
safety k = ∀ {A} (E : Environment A) → Invariant (protocol E) (safeState k E)

Invariant : (P : State → Type) → Type
Invariant P = (s₁ s₂ : State) → Trace s₁ s₂ → P s₁ → P s₂
\end{minted}
  \begin{itemize}
    \item Cutting the last \agda{k} blocks off one honest chain leaves a
          prefix of any other.
    \item \agda{protocol E = E ∘ (nodes ∘ᴷ network)}: the deployment,
          closed off by an environment that also plays the adversary.
    \item An invariant over every environment: traces preserve safe states.
    \item \agda{getChain} reads a chain through the node's
          \agda{IsBlockchain} queries.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Liveness: chain growth and chain quality}
\begin{minted}{agda}
hcgState : {A : Channel} → ℚ → (E : Environment A) → Machine.State (protocol E) → Type
hcgState τ E S₀ =
    {p : Fin n} (hp : p ∈ honest-nodes) {pref suff : List Block} {b : Block}
  → getChain E S₀ hp ≡ pref ++ (b ∷ suff)
  → isHonestBlock b
  → τ ℚ.* ℕ→ℚ (getSlot E S₀ hp ∸ slotOf b) ℚ.≤ ℕ→ℚ (length suff)

∃cqState : {A : Channel} → ℕ → (E : Environment A) → Machine.State (protocol E) → Type
∃cqState T E S₀ =
    {p : Fin n} (hp : p ∈ honest-nodes)
  → Any.Any isHonestBlock (recent T (getSlot E S₀ hp) (getChain E S₀ hp))
\end{minted}
  \begin{itemize}
    \item \textbf{Honest chain growth}: at least
          $\tau \cdot (\mathit{slot} - \mathit{slot}_b)$ blocks follow an
          honest block \agda{b}.
    \item \textbf{Existential chain quality}: among the blocks of the last
          \agda{T} slots of an honest chain, at least one is honest.
    \item \agda{hcg τ} and \agda{∃cq T} are invariants over every
          environment, like \agda{safety k}.
  \end{itemize}
\end{frame}

% Leios/Linear/Progress.agda: enough-traces, built from tick and upkeep.
\begin{frame}[fragile]{Progress: traces reach every future slot}
\begin{minted}{agda}
enough-traces : ∀ s n → allDone s
  → ∃[ s' ] slot s + n ≡ slot s' × Trace LinearLeios s s'
\end{minted}
  \begin{itemize}\small
    \item Safety and liveness are invariants along deployment traces.
          This progress result supplies bare-node witness traces;
          \agda{allDone} means all four per-slot duties are settled.
    \item \textbf{One tick}: \agda{Slot}, then \agda{Chain}, advance the
          slot and reset upkeep.  Settle EB production, base/certificate
          duties, then voting, restoring \agda{allDone} in the same slot.
    \item \textbf{No blocked duty}: produce or vote when eligible;
          otherwise use \agda{No-EB-Role} or \agda{No-VT-Role}.
          If the tip calls for a certificate, answer the node's query with
          \agda{CERT nothing} via \agda{Cert₁}.
    \item \textbf{Induct on \agda{n}}: the empty trace for zero;
          concatenate one tick with the remaining ticks using
          \agda{Trace-trans}.  Choose empty message and RB lists.
    \item \alert{Scope}: existential progress of the bare node, with
          inputs chosen by the trace; not chain growth or progress of the
          composed deployment, nor a premise of the transfer theorems.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Safety and liveness transfer}
  \begin{itemize}
    \item \textbf{Safety transfer}: assuming the chain lemma, Leios
          inherits common prefix from the base chain.
  \end{itemize}
\begin{minted}{agda}
leiosSafety : (∀ {A} (E : S.Environment A) → TrM.ChainLemma-ty E)
            → Deployment.safety Tr.base k → S.safety k
\end{minted}
  \begin{itemize}
    \item \textbf{Liveness transfer}: honest chain growth and chain quality
          (\agda{leiosHCG}, \agda{leios∃CQ}), from chain and slot lemmas.
    \item The key step is \agda{is-extension}: the Leios node \emph{is}
          the base node with an extra layer on top, up to \agda{≅ᴹ}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{When are two machines the same?}
\begin{minted}{agda}
record _≅ᴹ_ (M M' : Machine A B) : Type where
  field to      : S → S'
	from    : S' → S
        from∘to : ∀ s  → from (to s) ≡ s
	to∘from : ∀ s' → to (from s') ≡ s'
        step-to   : R  s₁  i o s₂  → R' (to s₁)    i o (to s₂)
        step-from : R' s₁' i o s₂' → R  (from s₁') i o (from s₂')
\end{minted}
  \begin{itemize}
    \item \agda{≅ᴹ}: a state bijection that preserves and reflects steps.
    \item Up to \agda{≅ᴹ}, machines with \agda{∘}, \agda{⊗₁}, \agda{I} form
          a \alert{symmetric monoidal category}, all laws proved.
    \item Strong, and sometimes too strong: a bisimulation \agda{\_≈ᴮ\_}
          is available where state spaces differ.
  \end{itemize}
\end{frame}

% Blockchain/Safety/Transfer.agda; Blockchain/Liveness/Transfer.agda reuses
% the same transProtocol.
\begin{frame}[fragile]{The category at work: transferring safety and liveness}
  \begin{itemize}
    \item Channels are the objects, machines the morphisms; with \agda{∘},
          \agda{⊗₁} and \agda{I} they form a symmetric monoidal category up
          to \agda{≅ᴹ}.  The Kleisli operators \agda{∘ᴷ}, \agda{⊗ᴷ}
          additionally carry the adversary channels along.
    \item The transfer proofs rewrite the closed Leios system, for any
          environment \agda{E}, into the \emph{base} system under a new
          environment:
  \end{itemize}
\begin{minted}{agda}
transProtocol : Ext.protocol E ≅ᴹ Base.protocol transEnv
transEnv      = E ∘ transId ∘ ⨂ᴷ extPart ⊗₁ id
\end{minted}
  \begin{itemize}
    \item Each honest Leios node is its Leios layer on top of a base node
          (\agda{is-extension}); the n-ary laws \agda{⨂-reshape-env} and
          \agda{⨂-absorb-env}, derived from the monoidal ones, then move all
          the layers into the environment.
    \item \agda{≅ᴹ} maps states and traces
          (\agda{Trace-map transProtocol}).  Base safety and liveness hold
          for \emph{every} environment, so also for \agda{transEnv}; with
          the chain lemma (base chain $=$ Leios chain mapped to RBs), and
          for liveness the slot lemma, they carry over to Leios.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Building a channel with \agda{simpleChannel}}
\begin{minted}{agda}
data Mode : Type where
  Out In : Mode

simpleChannel : (Mode → Type) → Channel
simpleChannel T = T In ⇿ T Out

modeType : Mode → Channel → Type
modeType Out = outType
modeType In  = inType
\end{minted}
  \begin{itemize}
    \item Define one message family \agda{T : Mode → Type}, indexing each
          constructor by its direction, \agda{In} or \agda{Out}.
    \item \agda{simpleChannel T} packages the two types into a channel:
          \agda{inType = T In} and \agda{outType = T Out}.
    \item \agda{modeType} selects a direction's message type from a channel;
          \agda{¬ₘ} swaps \agda{In} and \agda{Out}.  These let the selection
          DSL track directions through bundles and transposes.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{The selection DSL}
Messages into a bundle are addressed with a selection DSL.
\begin{minted}[fontsize=\scriptsize]{agda}
data _[_]⇒[_]ᵍ_ : Channel → Mode → Mode → Channel → Type₁ where
  ϵ   : A [ m ]⇒[ m ]ᵍ A
  _⊗R : A [ m ]⇒[ m' ]ᵍ B → A [ m ]⇒[ m' ]ᵍ B ⊗₀ C
  L⊗_ : A [ m ]⇒[ m' ]ᵍ B → A [ m ]⇒[ m' ]ᵍ C ⊗₀ B
  _ᵗ¹ : A [ m ]⇒[ m' ]ᵍ B → A [ m ]⇒[ ¬ₘ m' ]ᵍ B ᵀ
  _ᵗ² : A [ m ]⇒[ ¬ₘ m' ]ᵍ B → A [ m ]⇒[ m' ]ᵍ B ᵀ

_↑ᵢ_ : A [ In ]⇒[ m' ]ᵍ B → modeType In A → modeType m' B
_↑ₒ_ : A [ Out ]⇒[ m' ]ᵍ B → modeType Out A → modeType m' B
\end{minted}
  \begin{itemize}
    \item A path locates channel \agda{A} inside a bundle \agda{B};
          \agda{↑ᵢ} and \agda{↑ₒ} inject an input or output message of
          \agda{A} along it.
    \item Read it from the inside out: \agda{ϵ} is the message's own
          channel, \agda{\_⊗R} puts it on the left of a \agda{⊗₀},
          \agda{L⊗\_} on the right, and \agda{ᵗ¹}, \agda{ᵗ²} cross a
          \agda{ᵀ}, which flips the direction.
  \end{itemize}
\end{frame}

          this selection.  Inclusion does not remove transactions from it;
          the environment must update the selection.
    \item \agda{FetchLdgI} requests the Leios ledger;
          \agda{FetchLdgO txs} returns the node's \agda{Ledger}, without
          changing its state.  This is distinct from the RB chain fetched
          through \agda{BaseIO}.
    \item The transpose \agda{ᵀ} makes the environment's requests inputs
          to the node at its codomain interface.
  \end{itemize}
\end{frame}

% Leios/Protocol.lagda.md: Types.Adv.
\begin{frame}[fragile]{\agda{Adv}: an empty adversary interface}
\begin{minted}{agda}
Adv : Channel
Adv = I

-- The empty channel has no messages in either direction.
I .Channel.inType  = ⊥
I .Channel.outType = ⊥
\end{minted}
  \begin{itemize}
    \item \agda{LinearLeios} exposes no direct adversarial commands or
          replies.  Its \agda{Adv} wire is the empty channel, the tensor unit.
    \item This does not remove the adversary from the deployment: the
          diffusion network, voting certifier and base chain have their own
          adversary interfaces.
    \item Corrupt parties are controlled through the deployment's wiring;
          the honest-node machine specifies the behaviour of honest parties.
    \item Keeping \agda{Adv} explicit gives the node the same compositional
          shape as components with nonempty adversary interfaces.
  \end{itemize}
\end{frame}
